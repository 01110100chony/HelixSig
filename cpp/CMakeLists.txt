cmake_minimum_required(VERSION 3.20)
project(helix_kernel VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
option(HELIX_SANITIZERS "Instrument standalone native tests with ASan/UBSan" OFF)
option(HELIX_BENCHMARKS "Build the standalone H5 measurement reference" OFF)

add_library(helix_kernel STATIC src/kernel.cpp)
target_include_directories(helix_kernel PUBLIC include)
target_compile_features(helix_kernel PUBLIC cxx_std_20)
set_target_properties(helix_kernel PROPERTIES CXX_EXTENSIONS OFF)
target_compile_options(helix_kernel PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)

if(HELIX_SANITIZERS)
  target_compile_options(helix_kernel PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(helix_kernel PUBLIC -fsanitize=address,undefined)
endif()

if(BUILD_TESTING)
  add_executable(kernel_tests tests/kernel_tests.cpp)
  target_link_libraries(kernel_tests PRIVATE helix_kernel)
  add_test(NAME kernel_contract COMMAND kernel_tests)
  set_tests_properties(kernel_contract PROPERTIES TIMEOUT 15)
  add_executable(kernel_driver tools/kernel_driver.cpp)
  target_link_libraries(kernel_driver PRIVATE helix_kernel)
endif()
if(HELIX_BENCHMARKS)
  add_executable(kernel_bench tools/kernel_bench.cpp)
  target_link_libraries(kernel_bench PRIVATE helix_kernel)
  target_compile_options(kernel_bench PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
endif()
install(TARGETS helix_kernel ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
